# Boots FlorentRevest/u-boot (arm64) instead of a Linux kernel.
arm_64bit=1
kernel=u-boot.bin

# u-boot's own console (CONFIG_DM_VIDEO=y, real bcm2835_video/HDMI driver -
# confirmed compiled into precompiled/u-boot.bin via `strings`) needs the
# GPU to actually bring HDMI up rather than skip it because nothing
# responded to hotplug detection yet.
hdmi_force_hotplug=1
disable_overscan=1

# Serial fallback in case the HDMI framebuffer handoff to XNU (see
# README.md) doesn't work as reasoned - GPIO14/15 @ 115200 8N1, needs a
# USB-serial adapter into the header, not the HDMI cable. Harmless to leave
# on either way.
enable_uart=1

# Real Pi 3 mini-UART (BCM2837_AUX_MU_*, what both u-boot's serial console
# and pexpert/arm/pe_serial.c's PI3_UART driver use on GPIO14/15) derives
# its baud rate from the VPU core clock, NOT a fixed UART clock like the
# PL011 has - its divisor register (baudreg=270 in pe_serial.c) is a fixed
# value computed assuming a 250MHz core clock (270 -> ~115200 baud at
# 250MHz). Without pinning core_freq, the firmware is free to scale that
# clock (e.g. up under HDMI/GPU load), which silently shifts the ACTUAL
# baud rate away from 115200 while every -b setting on the host side still
# says 115200 - the real, confirmed (real-hardware boot test, 2026-08-20)
# symptom is garbled-but-present serial output, not blank. Locking the
# clock is the standard rpi3 mini-UART fix, not a guessed alternate baud
# value.
core_freq=250

# The VideoCore's share of RAM. The firmware keeps it at the top of the 1 GB;
# u-boot's bootxnu hands XNU the rest (boot_args->mem_size from the
# firmware's GET_ARM_MEMORY answer, third_party/u-boot/cmd/bootxnu.c - it
# used to be a fixed 128 MB, which an older version of this comment said).
# 128 MB rather than 64: the VPU's H.264 decoder (MMAL vc.ril.video_decode,
# reached through BCM2837VCHIQ for Venice's hardware decoding) allocates its
# reference frames and output buffers from this memory, and 64 MB is not
# enough for 1080p; 128 is what Raspberry Pi OS sets for video playback.
# XNU gets 896 MB instead of 960.
gpu_mem=128

# Run the ARM cores at their rated 1.2 GHz all the time. The firmware only
# raises the ARM clock above its 600 MHz floor when the OS asks (Linux's
# cpufreq driver does, through the mailbox's set-clock-rate tag); nothing in
# this port does, so the Pi ran at half speed. Measured on a real Pi 3
# (2026-10-17): a dependent-add loop ran exactly twice as fast with this set,
# and H.264 720p decoding went from ~30 to ~60 frames/s (Venice dropped 9 of
# 300 frames instead of 176). force_turbo only disables dynamic clocking; it
# does not overvolt (no over_voltage here), and the firmware's own thermal
# throttling still applies. core_freq above stays 250 for the mini-UART:
# with it set explicitly, that is also its turbo value. A cpufreq-style
# driver that asks for the clock (and could lower it when idle) would be the
# finer-grained alternative.
force_turbo=1
